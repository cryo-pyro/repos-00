    """
Ukubona LLC — Gastric Toy Twin Simulation
==========================================
A minimal viable "digital twin" for antral gastritis,
demonstrating multi-objective loss optimization and
Bayesian uncertainty quantification.

Patient state space:
    x = [Gastric Inflammation, Lower GI Pathology, Pain Signal, Mucosal Integrity]

Loss functions:
    L_b  = biological cost (pain + inflammation)
    L_s  = system cost (procedure expense)
    L_total = L_b + α·L_s + β·H  (entropy-weighted)

Intervention actions:
    - PPI Trial     (low cost, high bio effect)
    - Endoscopy     (medium cost, high info gain for upper GI)
    - Colonoscopy   (high cost, near-zero info gain given symptom signal)

Run:
    python ukubona_toy_twin.py
"""

import numpy as np
import matplotlib.pyplot as plt
import matplotlib.patches as mpatches
from matplotlib.gridspec import GridSpec
from scipy.stats import entropy as scipy_entropy

np.random.seed(42)

# ─────────────────────────────────────────────
# 1. STATE SPACE
# ─────────────────────────────────────────────
# Dimensions: [Gastric Inflammation I, Lower GI Pathology G,
#              Pain Signal P, Mucosal Integrity M]
# Values ∈ [0, 1].  Higher = worse, except M (higher = healthier).

INITIAL_STATE = np.array([0.82, 0.04, 0.75, 0.38])
# Interpretation: high gastric inflammation, negligible lower GI signal,
# significant pain, compromised mucosa.

# ─────────────────────────────────────────────
# 2. LOSS FUNCTIONS
# ─────────────────────────────────────────────
ALPHA = 0.20   # weight on system/economic loss
BETA  = 0.10   # weight on entropy (uncertainty cost)

def L_bio(state):
    I, G, P, M = state
    return 0.40 * I + 0.40 * P + 0.20 * (1 - M)

def L_sys(cost):
    return cost

def H_state(state):
    """Entropy proxy: variance across biological dimensions."""
    probs = np.abs(state) / (np.sum(np.abs(state)) + 1e-9)
    return float(scipy_entropy(probs + 1e-9))

def L_total(state, cost):
    return L_bio(state) + ALPHA * L_sys(cost) + BETA * H_state(state)

# ─────────────────────────────────────────────
# 3. ACTIONS
# ─────────────────────────────────────────────
ACTIONS = {
    "PPI Trial": {
        "cost": 0.05,
        "delta": np.array([-0.30, 0.00, -0.38, +0.22]),   # reduces I, P; improves M
        "voi":   0.35,   # moderate info gain
        "label": "Apollonian (Optimal)"
    },
    "Endoscopy": {
        "cost": 0.40,
        "delta": np.array([-0.05, 0.00, -0.05, +0.05]),   # confirms, minor benefit
        "voi":   0.88,   # high info gain for upper GI
        "label": "Apollonian (Diagnostic)"
    },
    "Colonoscopy": {
        "cost": 0.80,
        "delta": np.array([0.00, 0.00, 0.00, -0.03]),     # no therapeutic effect
        "voi":   0.02,   # near-zero info gain
        "label": "Dionysian Error"
    },
    "Profit-Optimised": {
        "cost": 0.80,
        "delta": np.array([0.00, 0.00, 0.00, -0.03]),
        "voi":   0.02,
        "label": "Hidden Loss (System)"
    },
}

# ─────────────────────────────────────────────
# 4. STATE TRANSITION (simple linear ODE step + noise)
# ─────────────────────────────────────────────
HORIZON = 12   # time steps (e.g., days / visits)
SIGMA_PROC = 0.025

def transition(state, delta, sigma=SIGMA_PROC):
    noise = np.random.normal(0, sigma, size=state.shape)
    return np.clip(state + delta + noise, 0.0, 1.0)

# ─────────────────────────────────────────────
# 5. SIMULATION PATHS
# ─────────────────────────────────────────────
def simulate_path(initial_state, action_key, horizon=HORIZON):
    action = ACTIONS[action_key]
    state  = initial_state.copy()
    losses = [L_total(state, 0)]
    lb_arr = [L_bio(state)]
    H_arr  = [H_state(state)]
    states = [state.copy()]

    for t in range(horizon):
        # Apply action effect with decaying magnitude
        decay  = np.exp(-0.25 * t)
        effective_delta = action["delta"] * decay
        cost   = action["cost"] if t == 0 else 0.0
        state  = transition(state, effective_delta)
        losses.append(L_total(state, cost))
        lb_arr.append(L_bio(state))
        H_arr.append(H_state(state))
        states.append(state.copy())

    return np.array(losses), np.array(lb_arr), np.array(H_arr), np.array(states)

paths = {}
for k in ACTIONS:
    paths[k] = simulate_path(INITIAL_STATE, k)

# ─────────────────────────────────────────────
# 6. BAYESIAN POSTERIOR (symptom-constrained)
# ─────────────────────────────────────────────

    """Value of Information based on symptom gradient alignment."""
    return ACTIONS[action_key]["voi"]
