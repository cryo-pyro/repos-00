```sql
-- Ukubona-Strata Schema: Titan Substrate to Olympian Ledger
-- Created: February 24, 2026 (Kampala Time) for Strata Stone Partners
-- Mythic Mapping: Enclosing Gaia (Mass) within Uranus (Boundaries), filtering to Apollo (Client Face)

-- Enable extensions for timestamps and UUIDs if needed
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Supporting: Users (Actors - CEO, Board, Agents, now with Client role emphasized)
CREATE TABLE users (
    user_id BIGSERIAL PRIMARY KEY,
    role ENUM('ceo', 'board', 'agent', 'client') NOT NULL,
    location VARCHAR(255) DEFAULT 'Kampala',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Clients: The "Face" - Future Homeowners as Oracle
-- Expanded for residential focus: project_type, vision_preferences (e.g., JSON for stone previews), eco_impact_preference
CREATE TABLE clients (
    client_id BIGSERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    location VARCHAR(255) DEFAULT 'South Dakota',
    investment_usd DECIMAL(15,2) DEFAULT 0.00,
    project_type ENUM('residential', 'commercial', 'infrastructure') DEFAULT 'residential',  -- Prioritizes homeowners
    vision_preferences JSONB,  -- e.g., {"stone_color": "gray", "finish": "polished"} for personalized prophecies
    eco_impact_preference ENUM('low', 'medium', 'high') DEFAULT 'medium',  -- Aligns with sustainability values
    user_id BIGINT REFERENCES users(user_id) ON DELETE SET NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Stone Blocks: Core Assets - Tracked Units from Quarry to Homeowner
CREATE TABLE stone_blocks (
    block_id BIGSERIAL PRIMARY KEY,
    rfid_tag VARCHAR(50) UNIQUE NOT NULL,
    origin_quarry VARCHAR(255) DEFAULT 'Kampala Quarry',
    client_id BIGINT REFERENCES clients(client_id) ON DELETE CASCADE,  -- Ties directly to "face" for momentum flow
    crypto_verification_hash VARCHAR(64),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Operations: Brute-Force Actions (Titan Work)
CREATE TABLE operations (
    operation_id BIGSERIAL PRIMARY KEY,
    ceo_agent_id BIGINT REFERENCES users(user_id),
    op_type ENUM('extraction', 'transport', 'cutting', 'maintenance') NOT NULL,
    start_time TIMESTAMP,
    end_time TIMESTAMP
);

-- MassNode: Extraction & Yield (Gaia Yield)
CREATE TABLE mass_yields (
    yield_id BIGSERIAL PRIMARY KEY,
    block_id BIGINT REFERENCES stone_blocks(block_id) ON DELETE CASCADE,
    extracted_tonnage DECIMAL(10,2) NOT NULL CHECK (extracted_tonnage >= 0),
    usable_volume_cbm DECIMAL(10,2) NOT NULL CHECK (usable_volume_cbm >= 0),
    waste_ratio DECIMAL(5,2) DEFAULT 0.0 CHECK (waste_ratio BETWEEN 0 AND 1),
    extraction_timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    instrument_method ENUM('drone_scan', 'weighbridge', 'laser') NOT NULL
);
CREATE INDEX idx_mass_block ON mass_yields(block_id);
CREATE INDEX idx_mass_time ON mass_yields(extraction_timestamp);

-- VelocityEdge: Flow & State-Change (Trajectory)
CREATE TABLE velocity_logs (
    log_id BIGSERIAL PRIMARY KEY,
    block_id BIGINT REFERENCES stone_blocks(block_id) ON DELETE CASCADE,
    current_location VARCHAR(255) NOT NULL,
    transit_time_minutes INT DEFAULT 0 CHECK (transit_time_minutes >= 0),
    customs_dwell_time_minutes INT DEFAULT 0 CHECK (customs_dwell_time_minutes >= 0),
    current_phase ENUM('raw', 'cut', 'polished', 'shipped', 'installed') NOT NULL,
    velocity_mps DECIMAL(10,4) DEFAULT 0.0,  -- Computed from GPS
    log_timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_velocity_block ON velocity_logs(block_id);
CREATE INDEX idx_velocity_time ON velocity_logs(log_timestamp);

-- FrictionSink: Waste Heat (Tartarus Dump)
CREATE TABLE friction_logs (
    friction_id BIGSERIAL PRIMARY KEY,
    block_id BIGINT REFERENCES stone_blocks(block_id) ON DELETE SET NULL,
    operation_id BIGINT REFERENCES operations(operation_id) ON DELETE CASCADE,
    idle_hours DECIMAL(5,2) DEFAULT 0.0 CHECK (idle_hours >= 0),
    maintenance_downtime_minutes INT DEFAULT 0 CHECK (maintenance_downtime_minutes >= 0),
    defect_rate DECIMAL(5,2) DEFAULT 0.0 CHECK (defect_rate BETWEEN 0 AND 1),
    route_delay_minutes INT DEFAULT 0 CHECK (route_delay_minutes >= 0),
    friction_type ENUM('equipment', 'logistics', 'quality', 'environmental') NOT NULL,
    friction_timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_friction_op ON friction_logs(operation_id);
CREATE INDEX idx_friction_time ON friction_logs(friction_timestamp);

-- TaxLedger: Capital Dissipation (Energy Tax)
CREATE TABLE capital_expenses (
    expense_id BIGSERIAL PRIMARY KEY,
    block_id BIGINT REFERENCES stone_blocks(block_id) ON DELETE SET NULL,
    operation_id BIGINT REFERENCES operations(operation_id) ON DELETE CASCADE,
    fuel_burn_rate_liters DECIMAL(10,2) DEFAULT 0.0 CHECK (fuel_burn_rate_liters >= 0),
    man_hours_logged DECIMAL(5,2) NOT NULL CHECK (man_hours_logged >= 0),
    consumables_cost_usd DECIMAL(10,2) DEFAULT 0.0 CHECK (consumables_cost_usd >= 0),
    total_energy_tax_usd DECIMAL(10,2) NOT NULL CHECK (total_energy_tax_usd >= 0),
    expense_timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_tax_op ON capital_expenses(operation_id);
CREATE INDEX idx_tax_time ON capital_expenses(expense_timestamp);

-- New: Client Prophecy Logs (Apollonian Face - Personalized Timelines for Homeowners)
CREATE TABLE client_prophecy_logs (
    prophecy_id BIGSERIAL PRIMARY KEY,
    client_id BIGINT REFERENCES clients(client_id) ON DELETE CASCADE,
    block_id BIGINT REFERENCES stone_blocks(block_id) ON DELETE CASCADE,
    predicted_timeline_days INT NOT NULL,  -- SGD-optimized delivery estimate
    current_status TEXT,  -- e.g., 'Phase 3: Polished - Eco Impact: Low'
    update_timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_prophecy_client ON client_prophecy_logs(client_id);

-- View: Momentum View (Computes p = m * v per block, aggregated)
CREATE VIEW momentum_view AS
SELECT 
    sb.block_id,
    c.client_id,
    AVG(my.usable_volume_cbm) AS avg_mass_m,
    AVG(vl.velocity_mps) AS avg_velocity_v,
    (AVG(my.usable_volume_cbm) * AVG(vl.velocity_mps)) AS momentum_p,
    SUM(ce.total_energy_tax_usd) AS energy_tax_usd,
    fl.defect_rate AS avg_defect_rate  -- For Tartarus filtering
FROM stone_blocks sb
JOIN clients c ON sb.client_id = c.client_id
JOIN mass_yields my ON sb.block_id = my.block_id
JOIN velocity_logs vl ON sb.block_id = vl.block_id
JOIN capital_expenses ce ON sb.block_id = ce.block_id
LEFT JOIN friction_logs fl ON sb.block_id = fl.block_id
GROUP BY sb.block_id, c.client_id, fl.defect_rate;

-- View: Optimization Dashboard (The Apollonian Metric: p / $ , filtered for Client Face)
CREATE VIEW optimization_dashboard AS
SELECT 
    block_id,
    client_id,
    momentum_p,
    energy_tax_usd,
    (momentum_p / NULLIF(energy_tax_usd, 0)) AS p_per_dollar,  -- Avoid div by zero
    avg_defect_rate
FROM momentum_view;

-- Trigger Example: Auto-Update Total Tax on Insert (Promethean Automation)
CREATE OR REPLACE FUNCTION update_total_tax() RETURNS TRIGGER AS $$
BEGIN
    NEW.total_energy_tax_usd := (NEW.fuel_burn_rate_liters * 1.5) + (NEW.man_hours_logged * 20) + NEW.consumables_cost_usd;  -- Placeholder calc (e.g., $1.5/liter fuel, $20/hr labor)
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trig_update_tax
BEFORE INSERT OR UPDATE ON capital_expenses
FOR EACH ROW EXECUTE PROCEDURE update_total_tax();
```
